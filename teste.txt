import cv2

for i in range(6):
    cap = cv2.VideoCapture(i, cv2.CAP_V4L2)

    if cap.isOpened():
        ok, frame = cap.read()
        if ok:
            print(f"/dev/video{i}: FUNCIONANDO! {frame.shape}")
        else:
            print(f"/dev/video{i}: Sem imagem")

    cap.release()
